add_library(AtreusPlatform OBJECT)
file(GLOB_RECURSE ATREUS_PLATFORM_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Public/*.h)
target_sources(AtreusPlatform
  PUBLIC
    FILE_SET HEADERS
    BASE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/Public
    FILES ${ATREUS_PLATFORM_HEADERS}
)

target_link_libraries(AtreusPlatform PUBLIC AtreusCommon)

target_compile_definitions(AtreusPlatform PRIVATE ATR_EXPORT)

file(GLOB_RECURSE ATREUS_PLATFORM_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Private/*.cpp)
file(GLOB_RECURSE ATREUS_PLATFORM_PRIVATE_HEADERS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/Private/*.h)

target_sources(AtreusPlatform PRIVATE ${ATREUS_PLATFORM_SOURCES} ${ATREUS_PLATFORM_PRIVATE_HEADERS})
target_include_directories(AtreusPlatform PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Private)

if(WIN32)
  set(ATREUS_PLATFORM_BACKEND_DIR ${CMAKE_SOURCE_DIR}/Platforms/Win32/Platform)
else()
  message(FATAL_ERROR "Atreus: AtreusPlatform has no backend implemented for this target platform")
endif()

file(GLOB ATREUS_PLATFORM_BACKEND_SOURCES CONFIGURE_DEPENDS ${ATREUS_PLATFORM_BACKEND_DIR}/Source/*.cpp)
file(GLOB ATREUS_PLATFORM_BACKEND_PRIVATE_HEADERS CONFIGURE_DEPENDS ${ATREUS_PLATFORM_BACKEND_DIR}/Source/*.h)

target_sources(AtreusPlatform PRIVATE ${ATREUS_PLATFORM_BACKEND_SOURCES} ${ATREUS_PLATFORM_BACKEND_PRIVATE_HEADERS})
target_include_directories(AtreusPlatform PRIVATE ${ATREUS_PLATFORM_BACKEND_DIR}/Source)

if(WIN32)
  target_link_libraries(AtreusPlatform PUBLIC user32)
endif()
